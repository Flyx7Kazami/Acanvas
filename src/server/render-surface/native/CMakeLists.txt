cmake_minimum_required(VERSION 3.21)
project(acanvas_surface LANGUAGES CXX OBJCXX)
option(ACANVAS_TEST_CLIPBOARD_TRIPWIRE "Instrument owned native test helper clipboard calls" OFF)
option(ACANVAS_TEST_CLIPBOARD_FAKE_HANDLE "Return a private owned pasteboard marker in native tests" OFF)
if(NOT APPLE)
  message(FATAL_ERROR "The current native HTML surface helper targets macOS.")
endif()
if(NOT CEF_ROOT)
  message(FATAL_ERROR "Set CEF_ROOT to the project CEF distribution directory.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(PROJECT_ARCH "arm64" CACHE STRING "CEF distribution architecture")
set(USE_SANDBOX ON CACHE BOOL "CEF renderer sandbox" FORCE)
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" "libcef_dll_wrapper")
set(CEF_TARGET acanvas-surface)
SET_CEF_TARGET_OUT_DIR()
set(CEF_APP "${CEF_TARGET_OUT_DIR}/${CEF_TARGET}.app")
set(EXECUTABLE_NAME "${CEF_TARGET}")
set(PRODUCT_NAME "${CEF_TARGET}")
add_executable(${CEF_TARGET} MACOSX_BUNDLE main.mm)
if(ACANVAS_TEST_CLIPBOARD_TRIPWIRE)
  target_compile_definitions(${CEF_TARGET} PRIVATE ACANVAS_TEST_CLIPBOARD_TRIPWIRE=1)
endif()
if(ACANVAS_TEST_CLIPBOARD_FAKE_HANDLE)
  target_compile_definitions(${CEF_TARGET} PRIVATE ACANVAS_TEST_CLIPBOARD_FAKE_HANDLE=1)
endif()
SET_EXECUTABLE_TARGET_PROPERTIES(${CEF_TARGET})
target_link_libraries(${CEF_TARGET} libcef_dll_wrapper ${CEF_STANDARD_LIBS})
set_target_properties(${CEF_TARGET} PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in")
set(FRAMEWORK_DIR "${CEF_APP}/Contents/Frameworks/Chromium Embedded Framework.framework")
add_custom_command(TARGET ${CEF_TARGET} POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E rm -rf "${FRAMEWORK_DIR}"
  COMMAND ${CMAKE_COMMAND} -E make_directory "${FRAMEWORK_DIR}/Versions/A"
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CEF_BINARY_DIR}/Chromium Embedded Framework.framework" "${FRAMEWORK_DIR}/Versions/A"
  COMMAND ${CMAKE_COMMAND} -E create_symlink "Versions/A/Chromium Embedded Framework" "${FRAMEWORK_DIR}/Chromium Embedded Framework"
  COMMAND ${CMAKE_COMMAND} -E create_symlink "Versions/A/Libraries" "${FRAMEWORK_DIR}/Libraries"
  COMMAND ${CMAKE_COMMAND} -E create_symlink "Versions/A/Resources" "${FRAMEWORK_DIR}/Resources"
  COMMAND ${CMAKE_COMMAND} -E create_symlink "A" "${FRAMEWORK_DIR}/Versions/Current"
  VERBATIM)
foreach(_suffix_list ${CEF_HELPER_APP_SUFFIXES})
  string(REPLACE ":" ";" _suffix_list ${_suffix_list})
  list(GET _suffix_list 0 _name_suffix)
  list(GET _suffix_list 1 _target_suffix)
  list(GET _suffix_list 2 _plist_suffix)
  set(_helper_target "acanvas-surface_Helper${_target_suffix}")
  set(_helper_output_name "acanvas-surface Helper${_name_suffix}")
  set(_helper_info_plist "${CMAKE_CURRENT_BINARY_DIR}/helper-Info${_target_suffix}.plist")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/helper-Info.plist.in" _plist_contents)
  string(REPLACE "\${EXECUTABLE_NAME}" "${_helper_output_name}" _plist_contents "${_plist_contents}")
  string(REPLACE "\${BUNDLE_ID_SUFFIX}" "${_plist_suffix}" _plist_contents "${_plist_contents}")
  file(WRITE "${_helper_info_plist}" "${_plist_contents}")
  add_executable(${_helper_target} MACOSX_BUNDLE helper.cc)
  SET_EXECUTABLE_TARGET_PROPERTIES(${_helper_target})
  target_link_libraries(${_helper_target} libcef_dll_wrapper ${CEF_STANDARD_LIBS})
  set_target_properties(${_helper_target} PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${_helper_info_plist}" OUTPUT_NAME "${_helper_output_name}")
  add_dependencies(${CEF_TARGET} ${_helper_target})
  add_custom_command(TARGET ${CEF_TARGET} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CEF_TARGET_OUT_DIR}/${_helper_output_name}.app" "${CEF_APP}/Contents/Frameworks/${_helper_output_name}.app" VERBATIM)
endforeach()
